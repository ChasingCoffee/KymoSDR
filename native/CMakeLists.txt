cmake_minimum_required(VERSION 3.24)
project(ThetisNative LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
option(THETIS_SANITIZE "Instrument native code with AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(THETIS_SANITIZE)
  if(MSVC)
    message(FATAL_ERROR "The combined sanitizer target currently requires Clang or GCC.")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
include(CTest)
find_package(Threads REQUIRED)
include(cmake/Dependencies.cmake)

set(WDSP_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../Project Files/Source/wdsp")
# The existing Windows project remains the authoritative DSP translation-unit
# list. Do not glob in experiments, or maintain a second copy of the algorithms.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${WDSP_SOURCE_DIR}/wdsp.vcxproj")
file(STRINGS "${WDSP_SOURCE_DIR}/wdsp.vcxproj" wdsp_entries REGEX "<ClCompile Include=\"")
foreach(entry IN LISTS wdsp_entries)
  string(REGEX REPLACE ".*Include=\"([^\"]+)\".*" "\\1" filename "${entry}")
  list(APPEND wdsp_sources "${WDSP_SOURCE_DIR}/${filename}")
endforeach()

add_library(wdsp SHARED ${wdsp_sources} interop/wdsp_abi.c)
target_include_directories(wdsp PRIVATE "${WDSP_SOURCE_DIR}" platform)
target_link_libraries(wdsp PRIVATE thetis_fftw3 thetis_rnnoise thetis_specbleach Threads::Threads)
if(WIN32)
  target_link_libraries(wdsp PRIVATE avrt)
  target_compile_definitions(wdsp PRIVATE _CRT_SECURE_NO_WARNINGS _USE_MATH_DEFINES)
else()
  target_sources(wdsp PRIVATE platform/wdsp_posix.c)
  target_link_libraries(wdsp PRIVATE m)
  target_compile_options(wdsp PRIVATE -Werror=implicit-function-declaration -Werror=incompatible-pointer-types)
endif()
set_target_properties(wdsp PROPERTIES
  OUTPUT_NAME thetis_wdsp
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/stage/$<CONFIG>"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/stage/$<CONFIG>")

if(BUILD_TESTING)
  target_sources(wdsp PRIVATE tests/wdsp_test_api.c)
  add_executable(wdsp_abi_tests tests/wdsp_abi_tests.c)
  target_link_libraries(wdsp_abi_tests PRIVATE wdsp)
  set_target_properties(wdsp_abi_tests PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/stage/$<CONFIG>")
  add_test(NAME wdsp_abi_and_noise COMMAND wdsp_abi_tests)
  set_tests_properties(wdsp_abi_and_noise PROPERTIES TIMEOUT 60)
  if(NOT WIN32)
    add_executable(platform_tests tests/platform_tests.c platform/wdsp_posix.c)
    target_include_directories(platform_tests PRIVATE platform)
    target_link_libraries(platform_tests PRIVATE Threads::Threads)
    add_test(NAME platform_primitives COMMAND platform_tests)
    set_tests_properties(platform_primitives PROPERTIES TIMEOUT 20)
  endif()
endif()
